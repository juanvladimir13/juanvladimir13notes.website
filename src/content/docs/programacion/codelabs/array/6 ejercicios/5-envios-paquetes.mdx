const destinationCities = ["La Paz", "Cochabamba", "Santa Cruz", "Sucre", "Tarija"];

const packages = [

## 4. Funciones de renderizado y cálculo acumulador

```javascript title="app.js" showLineNumbers
// Renderizar ciudades de destino primitivas
function mostrarCiudades() {
  let texto = "";
  for (let i = 0; i < destinationCities.length; i++) {
    texto += `📍 ${destinationCities[i]}  `;
  }
  outCiudades.textContent = texto;
}

// Renderizar la lista de objetos paquetes
function mostrarPaquetes() {
  let html = "";
  for (const pkg of packages) {
    const estado = pkg.delivered ? "📦 Entregado" : "🚚 En camino";
    html += `<li>[${pkg.tracking}] ${pkg.recipient} - ${pkg.weight} kg (${estado})</li>`;
  }
  outPaquetes.innerHTML = html;
}

// Acumular la suma total de pesos
function calcularPesoTotal() {
  let suma = 0;
  for (const pkg of packages) {
    suma += pkg.weight;
  }
  return suma;
}

// Actualizar propiedades en el array de objetos
function entregarTodos() {
  for (const pkg of packages) {
    pkg.delivered = true;
  }
}
```

## 5. Event Listeners

```javascript title="app.js" showLineNumbers
const btnMostrar = document.getElementById("btn-mostrar");
const btnPesoTotal = document.getElementById("btn-peso-total");
const btnEntregarTodos = document.getElementById("btn-entregar-todos");
const btnRegistrar = document.getElementById("btn-registrar");

btnMostrar.addEventListener("click", () => {
  mostrarPaquetes();
});

btnPesoTotal.addEventListener("click", () => {
  const carga = calcularPesoTotal();
  outPeso.textContent = `Carga total transportada: ${carga.toFixed(1)} kg`;
});

btnEntregarTodos.addEventListener("click", () => {
  entregarTodos();
  mostrarPaquetes();
});

// Registrar nuevo paquete
btnRegistrar.addEventListener("click", () => {
  const codigo = inpCodigo.value.trim();
  const destinatario = inpDestinatario.value.trim();
  const peso = Number(inpPeso.value);

  if (codigo === "" || destinatario === "" || !peso || peso <= 0) {
    alert("Por favor ingrese código, destinatario y un peso válido mayor a 0");
    return;
  }

  packages.push({
    tracking: codigo,
    recipient: destinatario,
    weight: peso,
    delivered: false
  });

  inpCodigo.value = "";
  inpDestinatario.value = "";
  inpPeso.value = "";

  mostrarPaquetes();
});

// Carga inicial
mostrarCiudades();
mostrarPaquetes();
```
