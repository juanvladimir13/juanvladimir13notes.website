## 4. Funciones de lógica y recorrido

Implementamos funciones para renderizar ambas estructuras de datos y realizar cálculos estadísticos:

```javascript showLineNumbers
function mostrarMaterias() {
  let texto = "";
  for (let i = 0; i < subjects.length; i++) {
    texto += `<span>${subjects[i]}</span>`;
  }
  outMaterias.innerHTML = texto;
}

function mostrarEstudiantes() {
  let html = "";
  for (const stu of students) {
    const condicion = stu.score >= 51 ? "Aprobado" : "Reprobado";
    const estado = stu.active ? "Activo" : "Inactivo";
    html += `<li>${stu.name} ${stu.score} (${condicion}) - ${estado}</li>`;
  }
  outEstudiantes.innerHTML = html;
}

// Calcular promedio general acumulando notas
function calcularPromedio() {
  if (students.length === 0) return 0;

  let suma = 0;
  for (const student of students) {
    suma += student.score;
  }
  return suma / students.length;
}

// Contar estudiantes aprobados (nota >= 51)
function contarAprobados() {
  let total = 0;
  for (const student of students) {
    if (student.score >= 51) {
      total++;
    }
  }
  return total;
}
```

## 5. Event Listeners
Vinculamos las acciones del usuario con los algoritmos creados:

```javascript showLineNumbers
const btnMostrar = document.getElementById("btn-mostrar");
const btnPromedio = document.getElementById("btn-promedio");
const btnAprobados = document.getElementById("btn-aprobados");
const btnAgregar = document.getElementById("btn-agregar");

// Botón para mostrar la lista completa
btnMostrar.addEventListener("click", () => {
  mostrarEstudiantes();
});

// Botón para calcular el promedio general
btnPromedio.addEventListener("click", () => {
  const promedio = calcularPromedio();
  outPromedio.textContent = `Promedio del grupo: ${promedio} pts.`;
});

// Botón para contar aprobados
btnAprobados.addEventListener("click", () => {
  const aprobados = contarAprobados();
  outAprobados.textContent = `Aprobados: ${aprobados} de ${students.length}`;
});

// Botón para registrar un nuevo estudiante
btnAgregar.addEventListener("click", () => {
  const nombre = inpNombre.value.trim();
  const nota = Number(inpNota.value);
  const activo = chkActivo.checked;

  if (nombre === "") {
    alert("Por favor ingrese un nombre y una nota válida entre 0 y 100");
    return;
  }

  // Insertamos el nuevo objeto al array
  students.push({ name: nombre, score: nota, active: activo });

  inpNombre.value = "";
  inpNota.value = "";
  chkActivo.checked = true;

  mostrarEstudiantes();
});

// Carga inicial de datos
mostrarMaterias();
mostrarEstudiantes();
```
