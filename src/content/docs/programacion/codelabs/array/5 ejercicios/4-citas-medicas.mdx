## 4. Funciones de renderizado y conteo condicional

```javascript title="app.js" showLineNumbers
// Mostrar lista de especialistas primitivos
function mostrarDoctores() {
  let texto = "";
  for (let i = 0; i < doctors.length; i++) {
    texto += `<span>${doctors[i]}</span>`;
  }
  outDoctores.innerHTML = texto;
}

// Renderizar array de objetos
function mostrarCitas() {
  let html = "";
  for (const cita of appointments) {
    const estado = cita.confirmed ? "✅ Confirmada" : "⏳ Pendiente";
    html += `<li>${cita.patient} (${cita.age} años) - ${estado}</li>`;
  }
  outCitas.innerHTML = html;
}

// Contar citas confirmadas
function contarConfirmadas() {
  let contador = 0;
  for (const cita of appointments) {
    if (cita.confirmed) {
      contador++;
    }
  }
  return contador;
}

// Contar pacientes pediátricos (menores de 18 años)
function contarPediatricos() {
  let totalMenores = 0;
  for (const cita of appointments) {
    if (cita.age < 18) {
      totalMenores++;
    }
  }
  return totalMenores;
}
```

## 5. Event Listeners

```javascript title="app.js" showLineNumbers
const btnMostrar = document.getElementById("btn-mostrar");
const btnConfirmadas = document.getElementById("btn-confirmadas");
const btnPediatricos = document.getElementById("btn-pediatricos");
const btnAgendar = document.getElementById("btn-agendar");

btnMostrar.addEventListener("click", () => {
  mostrarCitas();
});

btnConfirmadas.addEventListener("click", () => {
  const confirmadas = contarConfirmadas();
  outMetricas.textContent = `Citas: ${confirmadas} de ${appointments.length}`;
});

btnPediatricos.addEventListener("click", () => {
  const menores = contarPediatricos();
  outMetricas.textContent = `Pacientes menores: ${menores}`;
});

// Agendar nueva cita médica
btnAgendar.addEventListener("click", () => {
  const paciente = inpPaciente.value.trim();
  const edad = Number(inpEdad.value);
  const confirmada = chkConfirmada.checked;

  if (paciente === "") {
    alert("Por favor ingrese un paciente");
    return;
  }

  // Insertar nuevo objeto cita
  appointments.push({ patient: paciente, age: edad, confirmed: confirmada});

  inpPaciente.value = "";
  inpEdad.value = 0;
  chkConfirmada.checked = true;

  mostrarCitas();
});

// Inicializar interfaz
mostrarDoctores();
mostrarCitas();
```
