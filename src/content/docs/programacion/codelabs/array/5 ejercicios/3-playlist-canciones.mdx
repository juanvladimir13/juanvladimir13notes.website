const availableGenres = ["Rock", "Pop", "Electrónica", "Clásica"];

// 2. Array de objetos de canciones
const songs = [
  { title: "Bohemian", artist: "Queen", duration: 354, favorite: true },
  { title: "Thriller", artist: "Michael", duration: 294, favorite: false },
  { title: "California", artist: "Eagles", duration: 390, favorite: true }
];

## 4. Funciones de renderizado y acumuladores

```javascript title="app.js" showLineNumbers
// Renderizar géneros primitivos
function mostrarGeneros() {
  let texto = "";
  for (let i = 0; i < availableGenres.length; i++) {
    texto += `<span>${availableGenres[i]} </span>`;
  }
  outGeneros.innerHTML = texto;
}

// Renderizar lista de objetos canciones
function renderizarCanciones(lista) {
  let html = "";
  for (const song of lista) {
    const like = song.favorite ? " Like" : "Dislike";
    html += `<li>${song.title} - ${song.artist} ${like}</li>`;
  }
  outCanciones.innerHTML = html;
}

// Calcular la suma de duraciones en segundos
function calcularDuracionTotal() {
  let segundosTotales = 0;
  for (const song of songs) {
    segundosTotales += song.duration;
  }
  return segundosTotales;
}
```

## 5. Event Listeners

Conectamos la interactividad:

```javascript title="app.js" showLineNumbers
const btnTodas = document.getElementById("btn-todas");
const btnFavoritas = document.getElementById("btn-favoritas");
const btnDuracion = document.getElementById("btn-duracion");
const btnAgregar = document.getElementById("btn-agregar");

btnTodas.addEventListener("click", () => {
  renderizarCanciones(songs);
});

// Filtrar en un nuevo array las canciones con favorite === true
btnFavoritas.addEventListener("click", () => {
  const favoritas = [];
  for (const song of songs) {
    if (song.favorite) {
      favoritas.push(song);
    }
  }
  renderizarCanciones(favoritas);
});

// Mostrar duración acumulada
btnDuracion.addEventListener("click", () => {
  const totalSeg = calcularDuracionTotal();
  const min = Math.floor(totalSeg / 60);
  outDuracion.textContent = `Duración: ${min} min ${totalSeg} seg`;
});

// Registrar nueva canción
btnAgregar.addEventListener("click", () => {
  const titulo = inpTitulo.value.trim();
  const artista = inpArtista.value.trim();
  const segundos = Number(inpSegundos.value);
  const esFavorita = chkFavorita.checked;

  if (titulo === "") {
    alert("Por favor complete todos los campos correctamente");
    return;
  }

  songs.push({ title: titulo, artist: artista, duration: segundos, 
  favorite: esFavorita });

  inpTitulo.value = "";
  inpArtista.value = "";
  inpSegundos.value = 0;
  chkFavorita.checked = false;

  renderizarCanciones(songs);
});

// Inicio
mostrarGeneros();
renderizarCanciones(songs);
```
